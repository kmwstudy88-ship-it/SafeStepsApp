function Get-SafeStepsAdolescenceLesson {
    [CmdletBinding()]
    param(
        [Parameter(Mandatory = $true)]
        [ValidateSet('21','22','23','24','25')]
        [string]$LessonNumber
    )

    return $SafeStepsAdolescenceLessons |
        Where-Object { [string]$_.LessonNumber -eq $LessonNumber } |
        Select-Object -First 1
}

function Test-SafeStepsAdolescenceCompletion {
    [CmdletBinding()]
    param(
        [Parameter(Mandatory = $true)]
        [System.Collections.IDictionary]$Lesson,

        [Parameter(Mandatory = $true)]
        [System.Collections.IDictionary]$Progress
    )

    $reasons = [System.Collections.Generic.List[string]]::new()

    $definitionLength = 0
    if ($null -ne $Progress.DefinitionResponse) {
        $definitionLength = ([string]$Progress.DefinitionResponse).Trim().Length
    }

    if ($definitionLength -lt [int]$Lesson.DefinitionGate.MinimumCharacters) {
        $reasons.Add('Definition response is incomplete.')
    }

    if ([decimal]$Progress.VideoWatchPercent -lt [decimal]$Lesson.ProgressionRules.MandatoryVideoWatchPercent) {
        $reasons.Add('Required video watch percentage has not been reached.')
    }

    if (-not [bool]$Progress.AllKnowledgeChecksComplete) {
        $reasons.Add('Knowledge checks are incomplete.')
    }

    if (-not [bool]$Progress.HomePracticeTaskComplete) {
        $reasons.Add('Home practice or evidence task is incomplete.')
    }

    if (-not [bool]$Progress.FormalAssessmentComplete) {
        $reasons.Add('Formal assessment is incomplete.')
    }

    if ([decimal]$Progress.QuizPercent -lt [decimal]$Lesson.Quiz.PassPercent) {
        $reasons.Add('Quiz pass threshold has not been reached.')
    }

    if ([bool]$Lesson.ProgressionRules.CriticalSafetyQuestionsMustBeCorrect -and
        -not [bool]$Progress.CriticalSafetyQuestionsCorrect) {
        $reasons.Add('One or more critical safety questions must be corrected.')
    }

    if (-not [bool]$Progress.FinalReflectionComplete) {
        $reasons.Add('Final reflection is incomplete.')
    }

    [ordered]@{
        LessonNumber = [string]$Lesson.LessonNumber
        Complete = ($reasons.Count -eq 0)
        BlockingReasons = @($reasons)
    }
}

[ordered]@{
    Manifest = $SafeStepsAdolescenceCourseManifest
    Lessons = $SafeStepsAdolescenceLessons
