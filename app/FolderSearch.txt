# -----------------------------------------
# Find all SafeSteps folders on the system
# Output results to a readable text file
# -----------------------------------------

$OutputFile = "C:\SafeSteps_FolderSearch.txt"

Get-ChildItem -Path C:\ -Directory -Recurse -ErrorAction SilentlyContinue |
    Where-Object {
        $_.Name -match "safesteps" -or
        $_.FullName -match "safesteps"
    } |
    Select-Object FullName |
    Out-File -FilePath $OutputFile -Encoding UTF8

Write-Host "Search complete. Results saved to $OutputFile"
