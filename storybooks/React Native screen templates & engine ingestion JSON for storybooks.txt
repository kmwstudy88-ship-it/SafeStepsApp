// src/storybook/StoryScreen.tsx
import React from 'react';
import { View, Text } from 'react-native';
import { STORIES_1_TO_30, StoryDefinition, StoryStep } from './config/stories1to30';

interface StoryScreenProps {
  storyId: string;
  stepId: string;
}

export const StoryScreen: React.FC<StoryScreenProps> = ({ storyId, stepId }) => {
  const story: StoryDefinition | undefined = STORIES_1_TO_30.find(
    s => s.id === storyId,
  );
  if (!story) return <Text>Story not found</Text>;

  const step: StoryStep | undefined = story.uiFlow.find(
    st => st.id === stepId,
  );
  if (!step) return <Text>Step not found</Text>;

  return (
    <View style={{ flex: 1, padding: 16 }}>
      <Text style={{ fontSize: 20, fontWeight: '600' }}>{story.title}</Text>
      <Text style={{ marginVertical: 8 }}>{step.type.toUpperCase()}</Text>

      {/* Example dynamic component rendering; in production, map to real components */}
      {step.components.map(c => (
        <View key={c} style={{ marginVertical: 4 }}>
          <Text>{`[Component: ${c}]`}</Text>
        </View>
      ))}

      <View style={{ marginTop: 16 }}>
        <Text>{story.reflectionPrompt}</Text>
      </View>
    </View>
  );
};

// src/storybook/StoryNavigator.tsx
import React from 'react';
import { createNativeStackNavigator } from '@react-navigation/native-stack';
import { StoryScreen } from './StoryScreen';

const Stack = createNativeStackNavigator();

export const StorybookNavigator = () => (
  <Stack.Navigator screenOptions={{ headerShown: false }}>
    {Array.from({ length: 30 }).map((_, idx) => {
      const storyId = `storybook_${String(idx + 1).padStart(3, '0')}`;
      return (
        <Stack.Screen
          key={storyId}
          name={storyId}
          children={() => <StoryScreen storyId={storyId} stepId="intro" />}
        />
      );
    })}
  </Stack.Navigator>
);
